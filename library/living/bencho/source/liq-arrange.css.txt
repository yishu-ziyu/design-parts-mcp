/* 1 Â· free, and not fenced. The row goes wherever the finger
      goes with nothing clamping it â only its height decides

const x = (e.clientX - g.px) / k;

/* it stretches along whichever axis is moving and thins the
   other, then leans the way it is being thrown â a row flung
   sideways should not deform like one dropped straight down */
setWarp({
  x: clamp(Math.abs(dx) / dt / 2.6, 0, 0.26),
  y: clamp(Math.abs(dy) / dt / 2.6, 0, 0.26),
  tilt: clamp(dx / dt * 2.6, -7, 7),      // signed: a flick back rights it
});
transform: rotate(tilt) scale(1 + x - y * 0.55, 1 + y - x * 0.55);

/* no event arrives once the finger stops, so the warp would
   stay stuck at whatever it last was */
calm.current = setTimeout(() => setWarp({ x: 0, y: 0, tilt: 0 }), 90);

/* letting go of the position is what puts it back in line:
   the same element springs from wherever it was carried to
   the slot the others opened for it */
