/* The component is the button. A press does two things:
   the bell swings, and the button stays down. The
   confirmation is the control itself, so there is nothing to
   read and nothing to dismiss. */

/* The bell swings on a damped oscillation, not a single
   shake: each pass smaller than the last, which is what a
   struck bell actually does. */
bell.animate([
  { transform: "rotate(0deg)" },
  { transform: "rotate(-17deg)", offset: 0.11 },
  { transform: "rotate(14deg)",  offset: 0.27 },
  { transform: "rotate(-9deg)",  offset: 0.44 },
  { transform: "rotate(6deg)",   offset: 0.61 },
  { transform: "rotate(-3deg)",  offset: 0.78 },
  { transform: "rotate(0deg)" },
], { duration: 820, easing: "ease-out" });

/* it hangs from its crown, so that is where it pivots */
transform-origin: 50% 16%;

/* only the yes rings â turning it off is not an event the
   bell should celebrate */
if (!on) ring();

/* the label swap would jolt the pill from one width to
   another, so the two labels are stacked out of flow and the
   pill is given the width of whichever is showing. It grows
   into its answer instead of snapping to it. */
const shown = say.querySelector('[data-show="true"]');
setW(shown.offsetWidth);          // useLayoutEffect, pre-paint

.bell-say { transition: width 460ms cubic-bezier(0.24, 1.34, 0.38, 1); }
.bell-btn[data-on="true"] { background: var(--ink); color: var(--on-ink); }
